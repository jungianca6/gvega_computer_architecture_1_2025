//Instrucciones de memoria: Opcode 010 [bits 31-29], Operation [bit 28], Inmediato? [bit 27], RD [26-23], RS1 [22-19], Inmediato [18-0] 

LDR R0, R1                                      //Load (R1 + 0) -> R0           Operation 0, Inmediato 0        //Compilador pone el Inmediato 0 por defecto
LDR R0, R1, #Inmediato                          //Load (R1 + Inmediato) -> R0   Operation 0, Inmediato 0
LDRI R0, #inmediato                             //Load (Inmediato) -> R0        Operation 0, Inmediato 1

STR R0, R1                                      //Store (R1 + 0) -> R0          Operation 1, Inmediato 0        //Compilador pone el Inmediato 0 por defecto
STR R0, R1, #Inmediato                          //Store (R1 + Inmediato) -> R0  Operation 1, Inmediato 0
STRI R0, #inmediato                             //Store (Inmediato) -> R0       Operation 1, Inmediato 1
